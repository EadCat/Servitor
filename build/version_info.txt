#
# This is what "Properties > Details" shows when you right-click the built
# executable. PyInstaller reads it only on Windows and ignores it elsewhere,
# so keeping it referenced from the spec is safe on Linux.
#
# A template: build/Servitor.spec fills in the version from the VERSION file at
# the repository root and writes the result into the PyInstaller work directory.

    filevers=({version_tuple}),
    prodvers=({version_tuple}),
    mask=0x3f,

    date=(0, 0),
  ),
  kids=[

          StringStruct('ProductName', 'Servitor'),

      )
    ]),
    VarFileInfo([VarStruct('Translation', [1033, 1200])]),
  ],
